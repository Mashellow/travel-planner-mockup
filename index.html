// ฟังก์ชันประมวลผลคอนฟิกก่อนส่งไปประมวลผล Auto Plan Algorithm function prepareAutoPlanPayload(formData, modalData) { const { totalBudget, days, province } = formData; let activityBudget = 0; let accommodationBudget = 0; let anchorCoordinates = null; let autoFindHotel = false; switch (modalData.accommodationStatus) { case 'IN_APP_BOOKED': accommodationBudget = modalData.inAppBooking.price; activityBudget = totalBudget - accommodationBudget; anchorCoordinates = modalData.inAppBooking.coordinates; break; case 'EXTERNAL_BOOKED': anchorCoordinates = modalData.externalBooking.coordinates; // ได้จากการ Geocode ชื่อที่พัก if (modalData.externalBooking.isBudgetInclusive) { accommodationBudget = Number(modalData.externalBooking.hotelPrice) || 0; activityBudget = totalBudget - accommodationBudget; } else { activityBudget = totalBudget; // 100% } break; case 'NO_HOTEL': if (modalData.noHotel.wantRecommendation) { accommodationBudget = totalBudget * 0.35; // จัดสรรงบที่พัก 35% activityBudget = totalBudget * 0.65; // จัดสรรงบเที่ยว 65% autoFindHotel = true; } else { activityBudget = totalBudget; // 100% anchorCoordinates = getProvinceCenterZone(province); // ดึงพิกัดย่าน Center Zone } break; } return { province, days, totalBudget, activityBudget: Math.max(0, activityBudget), // ป้องกันงบติดลบ accommodationBudget, anchorCoordinates, autoFindHotel }; }
